// Helper de Áudio Universal para Web e Android (Capacitor)

class SoundManager {
  private unlocked = false;

  constructor() {
    // Desbloqueia o motor de áudio do Android no PRIMEIRO toque do utilizador no ecrã
    if (typeof window !== 'undefined') {
      const unlockAudio = () => {
        if (this.unlocked) return;
        
        // Cria um som silencioso para liberar o Web Audio API
        const silentAudio = new Audio('data:audio/wav;base64,UklGRigAAABXQVZFZm10IBIAAAABAAEARKwAAIhYAQACABAAAABkYXRhAgAAAAEA');
        silentAudio.play().then(() => {
          this.unlocked = true;
          document.removeEventListener('touchstart', unlockAudio);
          document.removeEventListener('click', unlockAudio);
        }).catch(() => {});
      };

      document.addEventListener('touchstart', unlockAudio, { once: true });
      document.addEventListener('click', unlockAudio, { once: true });
    }
  }

  // Função para tocar qualquer efeito sonoro
  public play(soundPath: string) {
    try {
      // Garante o caminho correto a partir da pasta public/
      const audio = new Audio(soundPath);
      
      // Força o canal de mídia do Android
      audio.volume = 1.0;
      
      const playPromise = audio.play();
      if (playPromise !== undefined) {
        playPromise.catch((error) => {
          console.warn("Autoplay do Android bloqueou o toque ou ficheiro não encontrado:", error);
        });
      }
    } catch (err) {
      console.error("Erro ao reproduzir som:", err);
    }
  }
}

export const soundManager = new SoundManager();
