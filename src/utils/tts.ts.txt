export const falarTexto = (texto: string) => {
  if (!('speechSynthesis' in window)) return;

  // Se estiver em pausa/bloqueado pelo Android, retoma
  window.speechSynthesis.resume();

  const utterance = new SpeechSynthesisUtterance(texto);
  utterance.lang = 'pt-PT'; // ou 'pt-BR'
  utterance.rate = 1.0;     // velocidade normal

  // Garantir que as vozes estão carregadas antes de falar
  const executarVoz = () => {
    const vozes = window.speechSynthesis.getVoices();
    const vozPT = vozes.find(v => v.lang.includes('pt'));
    if (vozPT) utterance.voice = vozPT;

    window.speechSynthesis.cancel(); // Limpa áudios anteriores encravados
    window.speechSynthesis.speak(utterance);
  };

  if (window.speechSynthesis.getVoices().length === 0) {
    window.speechSynthesis.onvoiceschanged = executarVoz;
  } else {
    executarVoz();
  }
};
